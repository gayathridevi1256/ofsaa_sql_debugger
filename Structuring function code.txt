  ONE_DAY_AMT, TWO_DAYS_AMT, TWO_CONSECUTIVE_FL, ONE_DAY_AMT_FL, TWO_DAYS_AMT_FL, FUNC_CRNCY_CD, REC_CT
 FROM
(select
     1 as Dataset_key,
     g.CUST_INTRL_ID
   , g.Trans_Dt
   , g.Bus_Day_Age
   , g.Clndr_Day_Age
   , 0 Grp_Cd
   , -1 as Trans_Ty
   , g.One_Day_Amt
   , g.One_Day_Amt + COALESCE(t.One_Day_Amt,0) as Two_Days_Amt
   , case when g.One_Day_Amt = (g.One_Day_Amt + COALESCE(t.One_Day_Amt,0)) then 0 else 1 end as Two_Consecutive_Fl
   -- it tells us if the first day fall into range
   , case when g.One_Day_Amt >= p_LE_R1 and g.One_Day_Amt <= p_UE_R1 then 1
          when g.One_Day_Amt >= p_LE_R2 and g.One_Day_Amt <= p_UE_R2 then 1
          else 0
      end as One_Day_Amt_Fl
   -- it tells us if sum of two consecutive days fall into range and we have two consecutive days
   , case when ((g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))>= p_LE_R1 and (g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))<= p_UE_R1 and g.One_Day_Amt <> (g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))) then 1
          when ((g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))>= p_LE_R2 and (g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))<= p_UE_R2 and g.One_Day_Amt <> (g.One_Day_Amt + COALESCE(t.One_Day_Amt,0))) then 1
          else 0
      end as Two_Days_Amt_Fl
  , g.Func_Crncy_Cd
  , count(1) over (partition by g.CUST_INTRL_ID) as Rec_Ct
from
      All_Trxn_B g
      left outer join All_Trxn_B t on g.CUST_INTRL_ID = t.CUST_INTRL_ID
                                   and g.bus_day_age = t.bus_day_age - 1
   ) a
WHERE
  --At least one attempt
  (a.One_Day_Amt_Fl <> 0
  OR
   a.Two_Days_Amt_Fl <> 0
  )
-- Min Number of Attempts
AND a.Rec_Ct >= p_Min_Attempts
ORDER BY a.CUST_INTRL_ID, a.Bus_Day_Age;

                 pLE_R1,
                 pLE_R2,
                 pUE_R1,
                 pUE_R2,
                 pMin_Attempts,
                 pIncld_Acct_Hldr_Typ_Cd)
   LOOP
   INSERT INTO fccmatomic.ML_STRUCTURING_CU_B_TMP
            VALUES (rec_cur.DATASET_KEY,

COMMIT;
